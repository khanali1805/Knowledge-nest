  191:   function reorderSections(sourceSectionId: string, targetSectionId: string) {
  192:     if (!selectedTheme || sourceSectionId === targetSectionId) {
  193:       return;
  194:     }
  195:     const sortedSections = selectedTheme.sections
  196:       .slice()
  197:       .sort((first, second) => first.position - second.position);

  199:       (section) => section.id === sourceSectionId,
  200:     );
  201:     const targetIndex = sortedSections.findIndex(
  202:       (section) => section.id === targetSectionId,
  203:     );

  205:       return;

  208:     sortedSections.splice(targetIndex, 0, movedSection);
